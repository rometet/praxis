cmake_minimum_required(VERSION 3.20)
project(PraxisMapTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(PraxisCompanionMapTests MapTests.cpp ../../src/map/MapCore.cpp ../../src/map/MapPersistence.cpp ../../src/map/chiyan/TerrainPalette.cpp ../../src/map/TerrainBaker.cpp)
target_include_directories(PraxisCompanionMapTests PRIVATE ../../src)
if(MSVC)
  target_compile_options(PraxisCompanionMapTests PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(PraxisCompanionMapTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(PraxisCompanionMapTests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME PraxisCompanionMapTests COMMAND PraxisCompanionMapTests)
